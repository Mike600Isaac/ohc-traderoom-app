<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Models\User;
use App\Services\KajabiService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class KajabiBridgeController extends Controller
{
    public function login(Request $request, KajabiService $kajabi)
    {
        $email = $request->query('email');
        $signature = $request->query('signature');

        // 1. Security Check: Verify the signature matches our Secret
        // This prevents people from just typing ?email=admin@ohc.com to hack in
        $expectedSignature = hash_hmac('sha256', $email, config('services.kajabi.secret'));

        if ($signature !== $expectedSignature) {
            return redirect('/login')->with('error', 'Invalid security signature.');
        }

        // 2. Find or Create the User (JIT Sync)
        $user = User::where('email', $email)->first();

        if (!$user) {
            $memberData = $kajabi->getKajabiMember($email);
            
            if (!$memberData) {
                return redirect('/login')->with('error', 'Member not found in Kajabi.');
            }

            $user = User::create([
                'kajabi_user_id' => $memberData['id'],
                'first_name'     => $memberData['first_name'],
                'last_name'      => $memberData['last_name'],
                'email'          => $email,
                'password'       => bcrypt(str()->random(16)),
                'current_path'   => 'Free', // We will update this in the next step
            ]);
        }

        // 3. Log them in
        Auth::login($user);

        return redirect()->route('dashboard');
    }
}
