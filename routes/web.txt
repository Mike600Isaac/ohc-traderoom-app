<?php

use App\Http\Controllers\ProfileController;
use Illuminate\Support\Facades\Route;
use App\Http\Controllers\Auth\KajabiBridgeController;

Route::get('/', [App\Http\Controllers\DashboardController::class, 'index'])
    ->middleware(['auth'])
    ->name('home');

Route::get('/auth/bridge/{kajabi_id}', function ($kajabi_id) {
    // In production, we will add a 'Secure Token' check here.
    // For now, we find the user by their Kajabi ID and log them in.
    $user = User::where('kajabi_user_id', $kajabi_id)->first();

    if ($user) {
        Auth::login($user);
        return redirect('/dashboard');
    }

    return redirect('/login')->with('error', 'Account not synced yet.');
});

Route::get('/auth/kajabi-bridge', [KajabiBridgeController::class, 'login'])->name('kajabi.bridge');

require __DIR__.'/auth.php';
